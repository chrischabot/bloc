# `bloc.ai`

REST mapping: [`/v1/ai`](../api/endpoints/ai.md). Required scope: `ai`.

## Types

```ts
type AISurface = 'writer' | 'ai_block' | 'agent' | 'autofill' | 'qa';
type AIModel   = 'default' | 'fast' | 'advanced';

interface AICompletionResponse {
  object:     'ai_completion';
  surface:    AISurface;
  model:      AIModel;
  text:       string;
  tokens_in:  number;
  tokens_out: number;
  citations:  Array<{ pageId: string; snippet: string; score: number }>;
}

interface AIAnswerResponse {
  object:  'ai_answer';
  answer:  string;
  sources: Array<{ page_id: string; snippet: string; score: number }>;
}

interface AIAgentResponse {
  object:  'agent_run';
  task_id: string;
  status:  'success' | 'partial' | 'failed';
  goal:    string;
  steps:   Array<{
    index:       number;
    type:        'tool_call' | 'llm';
    tool?:       string;
    input?:      Record<string, unknown>;
    output?:     unknown;
    message?:    string;
    status:      'success' | 'failed';
    duration_ms: number;
  }>;
  message: string;
}
```

## `bloc.ai.completions(args) → Promise<AICompletionResponse>`

```ts
args: {
  surface?:        AISurface;        // default 'writer'
  model?:          AIModel;          // default 'default'
  messages:        Array<{ role: 'system' | 'user' | 'assistant' | 'tool'; content: string; name?: string }>;
  context_pages?:  string[];         // page ids to retrieve context from
  block_id?:       string;           // with surface 'ai_block', persists into this block
}
```

## `bloc.ai.qa(args) → Promise<AIAnswerResponse>`

```ts
args: { query: string; filter?: { object?: 'page' | 'database' } }
```

Answers a workspace-grounded question with citations.

## `bloc.ai.autofillRun(args) → Promise<PropertyItem>`

```ts
args: { page_id: string; property_id: string; instructions?: string }
```

Returns the updated `PropertyItem`.

## `bloc.ai.agent(args) → Promise<AIAgentResponse>`

```ts
args: {
  goal:            string;
  max_iterations?: number;      // default 10
  context_pages?:  string[];
}
```

Runs a tool-using agent until completion or `max_iterations`. Synchronous — expect 5–30 s.
